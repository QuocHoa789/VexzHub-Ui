-- mk helper
local function mk(class, props, parent)
    local o = Instance.new(class)
    for k, v in pairs(props or {}) do o[k] = v end
    if parent then o.Parent = parent end
    return o
end

-- corner (ĐÃ FIX LỖI UDIM)
local function corner(p, r)
    if type(r) == "number" then
        r = UDim.new(0, r)
    elseif not r then
        r = UDim.new(0, 8)
    end
    return mk("UICorner", {CornerRadius = r}, p)
end

-- stroke
local function stroke(p, color, thickness, transparency)
    return mk("UIStroke", {
        Color = color or Theme.border,
        Thickness = thickness or 1,
        Transparency = transparency or 0,
    }, p)
end

-- gradient 3 keypoints
local function gradient3(p, mid, rot, transparencies)
    local g = mk("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, Theme.g1),
            ColorSequenceKeypoint.new(mid or 0.482699, Theme.g2),
            ColorSequenceKeypoint.new(1, Theme.g3),
        }),
        Rotation = rot or 0,
    }, p)
    if transparencies then
        g.Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, transparencies[1] or 0),
            NumberSequenceKeypoint.new(1, transparencies[2] or 0),
        })
    end
    return g
end

-- gradient custom keypoints
local function gradientCustom(p, points, rot, transparencies)
    local g = mk("UIGradient", {
        Color = ColorSequence.new(points),
        Rotation = rot or 0,
    }, p)
    if transparencies then
        g.Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, transparencies[1] or 0),
            NumberSequenceKeypoint.new(1, transparencies[2] or 0),
        })
    end
    return g
end

-- tween helper
local function tw(obj, goals, duration, style, direction)
    local t = TweenService:Create(obj,
        TweenInfo.new(duration or 0.3, style or Enum.EasingStyle.Quad, direction or Enum.EasingDirection.Out),
        goals)
    t:Play()
    return t
end

-- parse size
local function parseSize(str)
    if not str then return UDim2.new(0, 520, 0, 340) end
    local w, h = str:match("(%d+)%s*,%s*(%d+)")
    if w and h then return UDim2.new(0, tonumber(w), 0, tonumber(h)) end
    return UDim2.new(0, 520, 0, 340)
end

-- calc text height
local function calcTextHeight(text, textSize, font, maxWidth)
    local ok, size = pcall(function()
        return TextService:GetTextSize(text, textSize, font, Vector2.new(maxWidth, 10000))
    end)
    if ok and size then return size.Y end
    return 15
end

-- EffectClick2 closure

        BorderSizePixel = 0,

    g.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, Theme.g1),

        BackgroundTransparency = 1,
        Size = UDim2.new(0, 3, 0, 3),

    t:Play()
end

-- ResizeFrame closure
local function ResizeFrame(target, handle, ring)
    local resizing, startSize, startMouse = false, nil, nil
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1
           or i.UserInputType == Enum.UserInputType.Touch then
            resizing = true
            startSize = target.Size
            startMouse = i.Position
            if ring then
                task.spawn(function()
                    local rot = 0
                    while resizing do
                        rot = (rot + 10) % 360
                        ring.Rotation = rot
                        task.wait(0.02)
                    end
                end)
            end
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if resizing and (i.UserInputType == Enum.UserInputType.MouseMovement
                         or i.UserInputType == Enum.UserInputType.Touch) then
            local dx = i.Position.X - startMouse.X
            local dy = i.Position.Y - startMouse.Y
            target.Size = UDim2.new(0,
                math.clamp(startSize.X.Offset + dx, 420, 1000), 0,
                math.clamp(startSize.Y.Offset + dy, 300, 700))
        end
    end)
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1
           or i.UserInputType == Enum.UserInputType.Touch then
            resizing = false
        end
    end)
end

-- click closure
local function click(parent, text, icon, callback, hasDesc, desc)
    local h = (hasDesc or icon) and 55 or 35
    local f = mk("Frame", {
        BackgroundColor3 = Theme.text,
        BackgroundTransparency = 0,
        Size = UDim2.new(0.97, 0, 0, h),
        BorderSizePixel = 0,
        ClipsDescendants = true,
        Parent = parent,
    })
    corner(f, 10)
    gradient3(f, 0.5)

    local titleLabel = mk("TextLabel", {
        Name = "Title",
        BackgroundTransparency = 1,
        AnchorPoint = Vector2.new(0.5, hasDesc and 0.35 or 0.5),
        Position = UDim2.new(0.5, 0, hasDesc and 0.35 or 0.5, 0),
        Size = UDim2.new(0.95, 0, hasDesc and 0.4 or 0.8, 0),
        Text = text or "Button",
        TextColor3 = Theme.text,
        Font = Theme.fontBold,
        TextSize = 14,
        TextScaled = true,
        TextWrapped = true,
        Parent = f,
    })

    if hasDesc and desc then
        mk("TextLabel", {
            BackgroundTransparency = 1,
            AnchorPoint = Vector2.new(0.5, 0.7),
            Position = UDim2.new(0.5, 0, 0.7, 0),
            Size = UDim2.new(0.95, 0, 0.3, 0),
            Text = desc,
            TextColor3 = Theme.text,
            TextTransparency = 0.25,
            Font = Theme.fontReg,
            TextSize = 11,
            TextScaled = true,
            TextWrapped = true,
            Parent = f,
        })
    end

    local s = stroke(f, Theme.text, 2, 0.65)
    -- FIX ZINDEX
    local btn = mk("TextButton", {BackgroundTransparency = 1, Text = "", Size = UDim2.new(1, 0, 1, 0), ZIndex = 5, Parent = f})

    btn.MouseEnter:Connect(function()
        tw(f, {BackgroundTransparency = 0.4}, 0.3)
        tw(s, {Transparency = 0.85}, 0.3)
    end)
    btn.MouseLeave:Connect(function()
        tw(f, {BackgroundTransparency = 0}, 0.3)
        tw(s, {Transparency = 0.65}, 0.3)
    end)

    local function fire()
        EffectClick(_gui or f)
        if callback then
            task.spawn(function()
                local ok, err = pcall(callback)
                if not ok then warn("[AuzuraLib] click callback error:", err) end
            end)
        end
    end
    btn.MouseButton1Click:Connect(fire)

    return f, btn, titleLabel, fire
end

        if input.KeyCode == Enum.KeyCode.F2 then
            bg.Visible = not bg.Visible
            closeTxt.Text = bg.Visible and "Close" or "Open"
            if bg.Visible and firstOpen then

            end
        end
    end)

    -- mobile pinch
    local touches = {}
    local initialDist = nil
    local initialSize = nil
    UIS.TouchStarted:Connect(function(input)
        if #touches == 0 then
            touches[1] = input
        elseif #touches == 1 then
            touches[2] = input
            initialDist = (touches[1].Position - touches[2].Position).Magnitude
            initialSize = bg.Size
        end
    end)
    UIS.TouchMoved:Connect(function(input)
        if #touches >= 2 then
            for i, t in ipairs(touches) do
                if t == input then touches[i] = input; break end
            end
            if touches[1] and touches[2] then
                local dist = (touches[1].Position - touches[2].Position).Magnitude
                if initialDist and initialDist > 0 then
                    local scale = dist / initialDist
                    bg.Size = UDim2.new(0,
                        math.clamp(initialSize.X.Offset * scale, 420, 1000), 0,
                        math.clamp(initialSize.Y.Offset * scale, 300, 700))
                end
            end
        end
    end)
    UIS.TouchEnded:Connect(function(input)
        for i, t in ipairs(touches) do
            if t == input then table.remove(touches, i); break end
        end
        if #touches < 2 then initialDist = nil end
    end)

                Size = UDim2.new(0.5, 0, 0, 15),
                Text = c.Desc or "",
                TextColor3 = Theme.textDim,
                TextTransparency = 0.3,
                Font = Theme.fontReg,
                TextSize = 11,
                TextXAlignment = Enum.TextXAlignment.Left,
                TextScaled = true,
                TextWrapped = true,
                Parent = f,
            })

            local keyDisplay = mk("Frame", {
                Name = "KeyDisplay",
                BackgroundColor3 = Theme.text,
                BackgroundTransparency = 0.85,
                AnchorPoint = Vector2.new(1, 0),
                Position = UDim2.new(0.96, 0, 0, 8),
                Size = UDim2.new(0, 45, 0, 28),
                Parent = f,
            })
            corner(keyDisplay, 8)
            gradient3(keyDisplay, 0.482699, 45)

            mk("ImageLabel", {
                Name = "KeyIcon",
                BackgroundTransparency = 1,
                AnchorPoint = Vector2.new(0, 0.5),
                Position = UDim2.new(0.08, 0, 0.5, 0),
                Size = UDim2.new(0, 14, 0, 14),
                Image = "rbxassetid://10723407389",
                ImageColor3 = Theme.text,
                Parent = keyDisplay,
            })

            local keyText = mk("TextLabel", {
                Name = "KeyText",
                BackgroundTransparency = 1,
                AnchorPoint = Vector2.new(0.5, 0.5),
                Position = UDim2.new(0.6, 0, 0.5, 0),
                Size = UDim2.new(0.65, 0, 0.7, 0),
                Text = currentKey.Name,
                TextColor3 = Theme.text,
                Font = Theme.fontBold,
                TextSize = 13,
                TextScaled = true,
                Parent = keyDisplay,
            })

            -- FIX ZINDEX
            local modeBtn = mk("TextButton", {
                Name = "ModeButton",
                BackgroundColor3 = Color3.fromRGB(30, 30, 35),
                BackgroundTransparency = 0.3,
                AnchorPoint = Vector2.new(1, 0),
                Position = UDim2.new(0.96, 0, 0, 40),
                Size = UDim2.new(0, 70, 0, 20),
                Text = "",
                AutoButtonColor = false,
                ZIndex = 5,
                Parent = f,
            })
            corner(modeBtn, 6)
            local modeText = mk("TextLabel", {
                Name = "ModeText",
                BackgroundTransparency = 1,
                AnchorPoint = Vector2.new(0.5, 0.5),
                Position = UDim2.new(0.5, 0, 0.5, 0),
                Size = UDim2.new(0.9, 0, 0.8, 0),
                Text = currentMode,
                TextColor3 = Color3.fromRGB(200, 200, 220),
                Font = Theme.fontBold,
                TextSize = 10,
                TextScaled = true,
                Parent = modeBtn,
            })

            modeBtn.MouseButton1Click:Connect(function()
                currentMode = (currentMode == "Toggle") and "Hold" or "Toggle"
                modeText.Text = currentMode
                tw(modeBtn, {BackgroundTransparency = 0.1}, 0.2, Enum.EasingStyle.Back)
                task.wait(0.1)
                tw(modeBtn, {BackgroundTransparency = 0.3}, 0.2, Enum.EasingStyle.Back)
                EffectClick(bg)
            end)
            modeBtn.MouseEnter:Connect(function() tw(modeBtn, {BackgroundTransparency = 0.2}, 0.15) end)
            modeBtn.MouseLeave:Connect(function() tw(modeBtn, {BackgroundTransparency = 0.3}, 0.15) end)

            local listening = false
            -- FIX ZINDEX
            local clickBtn = mk("TextButton", {
                Name = "Click",
                BackgroundTransparency = 1,
                Text = "",
                Size = UDim2.new(1, 0, 1, 0),
                ZIndex = 5,
                Parent = f,
            })

            local function updateWidth(txt)
                local ok, ts = pcall(function()
                    return TextService:GetTextSize(txt, 13, keyText.Font, Vector2.new(1000, 1000))
                end)
                if ok and ts then
                    keyDisplay.Size = UDim2.new(0, math.clamp(ts.X + 24, 40, 60), 0, 28)
                end
            end

            clickBtn.MouseButton1Click:Connect(function()
                EffectClick(bg)
                listening = true
                keyText.Text = "..."
                task.wait(0.05)
                updateWidth("...")
            end)

            local connBegan, connEnded
            connBegan = UIS.InputBegan:Connect(function(input, gp)
                if listening then
                    if input.UserInputType == Enum.UserInputType.Keyboard then
                        currentKey = input.KeyCode
                        keyText.Text = currentKey.Name
                        listening = false
                        updateWidth(currentKey.Name)
                    end
                    return
                end
                if not gp and input.KeyCode == currentKey then
                    if currentMode == "Toggle" then
                        value = not value
                        if c.Callback then
                            task.spawn(function() pcall(c.Callback, currentKey, value, currentMode) end)
                        end
                    else
                        value = true
                        if c.Callback then
                            task.spawn(function() pcall(c.Callback, currentKey, true, currentMode) end)
                        end
                    end
                end
            end)
            connEnded = UIS.InputEnded:Connect(function(input)
                if input.KeyCode == currentKey and currentMode == "Hold" then
                    value = false
                    if c.Callback then
                        task.spawn(function() pcall(c.Callback, currentKey, false, currentMode) end)
                    end
                end
            end)
            f.Destroying:Connect(function()
                if connBegan then connBegan:Disconnect() end
                if connEnded then connEnded:Disconnect() end
            end)
            wireHighlight(f)

            return {
                Frame = f,
                SetKey = function(self, k) currentKey = k; keyText.Text = k.Name; updateWidth(k.Name) end,
                GetKey = function(self) return currentKey end,
                SetMode = function(self, m) currentMode = m; modeText.Text = m end,
                GetMode = function(self) return currentMode end,
                SetValue = function(self, v) value = v end,
                GetValue = function(self) return value end,
                SetTitle = function(self, t) f.Title.Text = t end,
                SetDesc = function(self, d) f.Desc.Text = d end,
                Visible = function(self, v) f.Visible = v end,
            }
        end

        -- ThumnailsImage
        function Tab:ThumnailsImage(c)
            c = c or {}
            local h = c.SizeY or 100
            local f = newCard(c.Side, h, "ImageFrame", true)
            local img = nil
            if c.Banner then
                img = mk("ImageLabel", {
                    Name = "Preview",
                    BackgroundTransparency = 1,
                    Size = UDim2.new(1, 0, 1, 0),
                    Image = "rbxassetid://" .. tostring(c.Banner),
                    ImageTransparency = 0.15,
                    ScaleType = Enum.ScaleType.Crop,
                    Parent = f,
                })
            end
            local titleLbl = nil
            if c.ShowTitle and c.Title then
                titleLbl = mk("TextLabel", {
                    Name = "ImageTitle",
                    BackgroundTransparency = 1,
                    AnchorPoint = Vector2.new(0.5, 0.5),
                    Position = UDim2.new(0.5, 0, 0.5, 0),
                    Size = UDim2.new(0.9, 0, 0.5, 0),
                    Text = c.Title,
                    TextColor3 = Theme.text,
                    Font = Theme.fontBold,
                    TextSize = 14,
                    TextScaled = true,
                    TextStrokeTransparency = 0.5,
                    Parent = f,
                })
            end

            local fullscreenFrame = mk("Frame", {
                Name = "FullscreenView",
                BackgroundColor3 = Color3.new(0, 0, 0),
                BackgroundTransparency = 1,
                Size = UDim2.new(1, 0, 1, 0),
                Visible = false,
                ZIndex = 200,
                Parent = gui,
            })

            local fullImg = nil
            if c.Banner then
                fullImg = mk("ImageLabel", {
                    BackgroundTransparency = 1,
                    AnchorPoint = Vector2.new(0.5, 0.5),
                    Position = UDim2.new(0.5, 0, 0.5, 0),
                    Size = UDim2.new(0, 0, 0, 0),
                    Image = "rbxassetid://" .. tostring(c.Banner),
                    ScaleType = Enum.ScaleType.Fit,
                    ZIndex = 201,
                    Parent = fullscreenFrame,
                })
            end

            local closeBtn = mk("TextButton", {
                BackgroundColor3 = Theme.bg,
                BackgroundTransparency = 0.1,
                Position = UDim2.new(1, -60, 0, 20),
                Size = UDim2.new(0, 40, 0, 40),
                Text = "X",
                TextColor3 = Theme.text,
                Font = Theme.fontBold,
                TextSize = 20,
                ZIndex = 202,
                Parent = fullscreenFrame,
            })
            corner(closeBtn, 8)
            stroke(closeBtn, Theme.border, 1, 0)

            local isFullscreen = false
            local function openFullscreen()
                if isFullscreen then return end
                isFullscreen = true
                fullscreenFrame.Visible = true
                tw(fullscreenFrame, {BackgroundTransparency = 0.3}, 0.3)
                if fullImg then
                    tw(fullImg, {Size = UDim2.new(0.8, 0, 0.8, 0)}, 0.3, Enum.EasingStyle.Back)
                end
            end
            local function closeFullscreen()
                if not isFullscreen then return end
                isFullscreen = false
                tw(fullscreenFrame, {BackgroundTransparency = 1}, 0.3)
                if fullImg then
                    tw(fullImg, {Size = UDim2.new(0, 0, 0, 0)}, 0.3, Enum.EasingStyle.Back)
                end
                task.delay(0.3, function()
                    if not isFullscreen then fullscreenFrame.Visible = false end
                end)
            end

            -- FIX ZINDEX
            local btn = mk("TextButton", {BackgroundTransparency = 1, Text = "", Size = UDim2.new(1, 0, 1, 0), ZIndex = 5, Parent = f})
            btn.MouseButton1Click:Connect(function()
                EffectClick(bg)
                openFullscreen()
            end)
            closeBtn.MouseButton1Click:Connect(function()
                EffectClick(gui)
                closeFullscreen()
            end)

            return {
                Frame = f,
                SetImage = function(self, id)
                    if img then img.Image = "rbxassetid://" .. tostring(id) end
                    if fullImg then fullImg.Image = "rbxassetid://" .. tostring(id) end
                end,
                SetTitle = function(self, t) if titleLbl then titleLbl.Text = t end end,
                SetHeight = function(self, h2)
                    tw(f, {Size = UDim2.new(0.97, 0, 0, h2)}, 0.3)
                end,
                Visible = function(self, v) f.Visible = v end,
            }
        end

        return Tab
    end
